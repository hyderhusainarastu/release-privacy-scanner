# Extra generic patterns for leakscan (--patterns patterns/default.txt).
# Format: '[CLASS]' or '[CLASS:WARN]' header, then one case-insensitive regex per line.
# The built-in classes (SECRET, PATH, EMAIL, HOSTNAME, AI_FINGERPRINT, IDENTITY) are
# already compiled in; this file adds shapes that are common but not universal.
# Keep it generic: owner-specific names belong in a private pattern file instead.

[INTERNAL_HOST:WARN]
\b[a-z0-9-]+\.(?:internal|corp|intranet)\b(?![.\w-])
\b(?:10\.\d{1,3}|192\.168|172\.(?:1[6-9]|2\d|3[01]))\.\d{1,3}\.\d{1,3}\b

[TICKET:WARN]
\b(?:INT|PRIV|CONF)-\d{2,6}\b

[DRAFT_MARKER:WARN]
\b(?:do not (?:publish|distribute|cite)|internal use only|confidential)\b
